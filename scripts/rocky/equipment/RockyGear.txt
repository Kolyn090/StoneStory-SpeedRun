// Strict preferred availability: never label a generic fallback as an element.
var swords = []
var wands = []
var hammers = []
var elements = ["poison", "vigor", "aether", "fire", "ice"]
var order = [3, 4, 1, 2, 0]
var physicalSword = ""
var healingSword = ""
var pickedElement = -1
var pickedKind = ""
var usedFallback = false
var missingPreferred = ""
var shield = ""
var shieldStatus = "not prepared"

func Available(selector)
  ?!string.Equals(selector, "")
    // Reject ambiguity instead of silently picking a possibly different variant.
    ?item.GetCount(selector) = 1
      return selector
  return ""

func Prepare(settings, profile)
  swords.Clear()
  wands.Clear()
  hammers.Clear()
  missingPreferred = ""
  shield = ""
  shieldStatus = "FastGenShl disabled; Towering family fallback"
  ?profile.fastGenShieldConfirmed
    shield = Available(profile.preferredShield)
    ?!string.Equals(shield, "")
      shieldStatus = "FastGenShl nickname; one native match"
    :
      shieldStatus = "FastGenShl missing/ambiguous; Towering family fallback"
  ?string.Equals(shield, "")
    shield = Available(profile.fallbackShield)
  ?string.Equals(shield, "")
    shieldStatus = "STOP: missing/ambiguous fallback shield"
  for i = 0..4
    swords.Add(Available(profile.swords[i]))
    wands.Add(Available(profile.wands[i]))
    hammers.Add(Available(profile.hammers[i]))
    ?string.Equals(swords[i], "")
      missingPreferred = missingPreferred + " S:" + elements[i]
    ?string.Equals(wands[i], "")
      missingPreferred = missingPreferred + " W:" + elements[i]
    ?string.Equals(hammers[i], "")
      missingPreferred = missingPreferred + " H:" + elements[i]
  physicalSword = Available(profile.physicalSword)
  healingSword = Available(settings.healingSword)
  ?string.Equals(physicalSword, "")
    missingPreferred = missingPreferred + " physical"
  ?string.Equals(missingPreferred, "")
    missingPreferred = "none"

func FromChoices(choices, preferred, blocked)
  ?preferred >= 0 & preferred < 5
    ?!blocked[preferred] & !string.Equals(choices[preferred], "")
      pickedElement = preferred
      return choices[preferred]
  for n = 0..4
    var index
    index = order[n]
    ?!blocked[index] & !string.Equals(choices[index], "")
      pickedElement = index
      return choices[index]
  return ""

func Pick(kind, preferred, blocked, physicalOnly)
  pickedElement = -1
  pickedKind = kind
  usedFallback = false
  ?physicalOnly
    pickedKind = "physical-sword"
    return physicalSword
  var choices
  var result
  choices = swords
  ?kind = "wand"
    choices = wands
  :?kind = "hammer"
    choices = hammers
  result = FromChoices(choices, preferred, blocked)
  ?!string.Equals(result, "")
    return result
  usedFallback = true
  ?kind = "hammer"
    result = FromChoices(swords, preferred, blocked)
    ?!string.Equals(result, "")
      pickedKind = "sword"
      return result
  ?kind ! "wand"
    pickedKind = "physical-sword"
    return physicalSword
  return ""
